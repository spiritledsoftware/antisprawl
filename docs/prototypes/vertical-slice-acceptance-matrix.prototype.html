export function gatherRunnableJobs(workItems: Job[]): Job[] {
  const runnableJobs: Job[] = [];
  for (const item of workItems) {
    if (item.status !== "ready") continue;
    if (item.cancelledAt != null) continue;
    if (item.paused) continue;
    runnableJobs.push(item);
  }
  return runnableJobs;
}

export function collectOverdueInvoices(
  invoices: Invoice[], today: number,
): Invoice[] {
  const overdue: Invoice[] = [];
  for (const invoice of invoices) {
    if (invoice.status !== "open") continue;
    if (invoice.dueAt >= today) continue;
    overdue.push(invoice);
  }
  return overdue;
}
